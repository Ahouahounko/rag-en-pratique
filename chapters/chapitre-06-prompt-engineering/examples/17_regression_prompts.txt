# Chaque entree memorise un cas et ce qu'on attend de lui.
# Les cas issus d'incidents reels sont les plus precieux :
# ils empechent la meme erreur de revenir.
JEU_DE_TEST = [
    {
        "question": "Quelle est la duree de la garantie ?",
        "extraits": ["cgv_p8"],
        "doit_contenir": ["24 mois"],
        "doit_citer": ["doc_1"],
        "origine": "cas nominal",
    },
    {
        "question": "La garantie est-elle transferable ?",
        "extraits": ["cgv_p8"],
        "doit_contenir": ["ne permettent pas de repondre"],
        "ne_doit_pas_contenir": ["generalement", "en principe"],
        "origine": "incident du 12/03 : reponse inventee",
    },
]


FONCTION evaluer_version(version_prompt) :

    resultats <- liste vide

    POUR CHAQUE cas DANS JEU_DE_TEST :

        reponse <- generer(version_prompt, cas.extraits, cas.question)

        succes <- VRAI
        POUR CHAQUE attendu DANS cas.doit_contenir :
            SI attendu ABSENT DE reponse ALORS succes <- FAUX

        POUR CHAQUE interdit DANS cas.ne_doit_pas_contenir :
            SI interdit PRESENT DANS reponse ALORS succes <- FAUX

        POUR CHAQUE source DANS cas.doit_citer :
            SI source ABSENTE DE reponse ALORS succes <- FAUX

        resultats.ajouter({cas, succes, reponse})

    # Le chiffre qui compte n'est pas le taux de reussite :
    # c'est la LISTE des cas qui passaient avant et echouent
    # maintenant. C'est cela, une regression.
    RETOURNER resultats
FIN FONCTION
