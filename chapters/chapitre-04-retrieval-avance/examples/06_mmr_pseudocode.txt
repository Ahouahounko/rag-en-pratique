ENTREE : candidats C, question q, nombre voulu k, lambda
SORTIE : selection S de k documents

S <- liste vide

# Le premier choix est libre : rien n'est encore selectionne,
# donc le terme de redondance est nul.
d* <- le document de C le plus similaire a q
deplacer d* de C vers S

TANT QUE taille(S) < k ET C n'est pas vide :

    POUR CHAQUE document d de C :

        pertinence <- similarite(d, q)

        # A quel point d ressemble-t-il a son plus proche
        # cousin DEJA retenu ? C'est le pire cas qui compte.
        redondance <- MAX sur s dans S de similarite(d, s)

        score(d) <- lambda * pertinence
                    - (1 - lambda) * redondance

    d* <- le document de C au meilleur score
    deplacer d* de C vers S

RETOURNER S
