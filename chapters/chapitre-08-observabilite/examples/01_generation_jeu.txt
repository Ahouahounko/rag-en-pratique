FONCTION generer_jeu(passages, taille, repartition) :

    jeu <- liste vide

    # --- Questions directes : un passage, une question ---
    POUR CHAQUE passage TIRE AU HASARD (taille * repartition.directe) :
        q <- modele("Ecris une question dont la reponse est
                     entierement contenue dans ce passage.
                     La question ne doit PAS citer le passage.",
                    passage)
        r <- modele("Reponds a cette question en utilisant
                     uniquement ce passage.", q, passage)
        jeu.ajouter({question: q, reponse: r,
                     passages_attendus: [passage.id],
                     type: "directe"})

    # --- Questions de synthese : deux passages du MEME document ---
    POUR CHAQUE couple (p1, p2) DU MEME DOCUMENT :
        q <- modele("Ecris une question qui exige les deux passages
                     pour etre traitee completement.", p1, p2)
        ...
        jeu.ajouter({..., passages_attendus: [p1.id, p2.id],
                     type: "synthese"})

    # --- Questions SANS REPONSE : l'etape que tout le monde saute ---
    # On les fabrique a partir de sujets VOISINS du corpus, pas de
    # sujets absurdes : "quelle est la capitale du Perou" ne teste
    # rien. "Quelle est la garantie du produit Z", quand seuls les
    # produits X et Y sont documentes, teste vraiment le refus.
    POUR CHAQUE sujet_voisin :
        q <- modele("Ecris une question plausible sur ce sujet,
                     dont la reponse n'est PAS dans le corpus.",
                    sujet_voisin)
        jeu.ajouter({question: q, reponse: "ABSTENTION_ATTENDUE",
                     passages_attendus: [], type: "sans_reponse"})

    RETOURNER jeu
FIN FONCTION
